# The board comes from the top-level CMakeLists (OPENDECK_BOARD, board.cmake).
idf_build_get_property(build_dir BUILD_DIR)
include(${build_dir}/opendeck_board.cmake)
set(srcs main.cpp gfx.cpp jpeg.cpp net.cpp ${OPENDECK_BOARD_DIR}/board.cpp)
if(BOARD_LINK STREQUAL "usb")
  list(APPEND srcs usb_link_tinyusb.cpp)
else()
  list(APPEND srcs usb_link_uart.cpp)
endif()
foreach(driver ${BOARD_DRIVERS})
  list(APPEND srcs drivers/${driver}.cpp)
endforeach()

# Local builds without the component manager pass the components in
# EXTRA_COMPONENT_DIRS and set OPENDECK_LOCAL_COMPONENTS; name them here.
set(local_requires mdns esp_littlefs)
if(BOARD_LINK STREQUAL "usb")
  list(APPEND local_requires esp_tinyusb tinyusb)
endif()
if(OPENDECK_BOARD STREQUAL "jc3248w535")
  list(APPEND local_requires esp_lcd_axs15231b esp_lcd_touch)
endif()
if(NOT DEFINED ENV{OPENDECK_LOCAL_COMPONENTS})
  set(local_requires)
endif()

idf_component_register(
  SRCS ${srcs}
  INCLUDE_DIRS "." "drivers" ${OPENDECK_BOARD_DIR}
  REQUIRES esp_lcd esp_driver_spi esp_driver_i2c esp_driver_ledc esp_driver_gpio esp_driver_uart
           esp_wifi esp_netif nvs_flash json mbedtls esp_timer esp_rom app_update ${local_requires})

# Default deck (generated by the app's exporter) baked into the storage partition.
littlefs_create_partition_image(storage ../fs_image/${BOARD_DECK} FLASH_IN_PROJECT)

target_compile_definitions(${COMPONENT_LIB} PRIVATE BOARD_ID="${OPENDECK_BOARD}")
